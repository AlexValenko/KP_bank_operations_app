21def write_down_report_default(func: callable) -> None: 

22 """Декоратор, записывает DataFrame в словарь и сохраняет его в файле data/default_reports.json""" 

23 

24 def wrapper(*args, **kwargs): 

25 reports_logger.debug("Вызван декоратор write_down_report_default") 

26 result = func(*args, **kwargs) 

27 if not isinstance(result, pd.DataFrame): 

28 reports_logger.error("Декорируемая функция должна возвращать DataFrame") 

29 raise TypeError("Функция должна возвращать DataFrame") 

30 

31 for col in result.columns: 

32 if pd.api.types.is_datetime64_any_dtype(result[col]): 

33 result[col] = result[col].dt.strftime("%Y-%m-%d %H:%M:%S") 

34 reports_logger.debug("Даты из DataFrame преобразованы в str для парсинга ...") 

35 

36 result_dict = result.to_dict("records") 

37 try: 

38 with open("data/default_reports.json", "w", encoding="utf-8") as file_report: 

39 json.dump(result_dict, file_report, indent=4, ensure_ascii=False) 

40 reports_logger.debug("Файл data/default_reports.json успешно записан \n") 

41 except Exception as e: 

42 print(f"Ошибка при сохранении отчета: {e}") 

43 reports_logger.error(f"Ошибка при сохранении отчета: {e}") 

44 

45 return result

84def spending_by_category(transactions: pd.DataFrame, category: str = "", date: Optional[str] = None) -> pd.DataFrame: 

85 """Функция возвращает траты по заданной категории за последние три месяца (от переданной даты).""" 

86 

87 # Создаем переменные для дат в datetime, чтобы фильтровать данные уже имеющейся функцией 

88 if date is None: 

89 end_date = datetime.datetime.today() 

90 else: 

91 try: 

92 end_date = datetime.datetime.strptime(date, "%d-%m-%Y") 

93 except (TypeError, ValueError) as e: 

94 reports_logger.error(f"Ошибка {e} чтения даты из аргументов функции, требуется dd-mm-YYYY") 

95 return pd.DataFrame() 

96 start_date = end_date - relativedelta(months=3) 

97 reports_logger.debug( 

98 f'Выбран диапазон дат с {start_date.strftime("%d-%m-%Y")} по {end_date.strftime("%d-%m-%Y")}' 

99 ) 

100 

101 filtered_by_date_df = filter_transactions_by_date(df=transactions, start_date=start_date, end_date=end_date) 

102 reports_logger.debug(f"Транзакции отфильтрованы по дате, выбрано {len(filtered_by_date_df)} записей") 

103 

104 # Фильтруем данные, чтобы остались только расходы (сумма операции < 0), статус - ОК, валюта - руб, категория есть 

105 cleaned_for_calc_df = filtered_by_date_df[ 

106 (filtered_by_date_df["Статус"] == "OK") 

107 & (filtered_by_date_df["Категория"].notna()) 

108 & (filtered_by_date_df["Сумма операции"] < 0) 

109 & (filtered_by_date_df["Валюта платежа"] == "RUB") 

110 ] 

111 

112 if cleaned_for_calc_df.empty: 

113 reports_logger.warning("Не найдено подходящих транзакций") 

114 return pd.DataFrame() 

115 reports_logger.debug(f"Транзакции отфильтрованы статусу и категории, выбрано {len(cleaned_for_calc_df)} записей") 

116 

117 grouped_by_category = cleaned_for_calc_df.groupby( 

118 "Категория", as_index=True, sort=True, group_keys=True, observed=True 

119 ).agg({"Сумма операции": "sum"}) 

120 reports_logger.debug(f"Транзакции сгруппированы по {len(grouped_by_category)} категориям") 

121 

122 if category in grouped_by_category.index: 

123 filtered_by_currency_category = cleaned_for_calc_df[cleaned_for_calc_df["Категория"] == category] 

124 reports_logger.debug(f"Найдено {len(filtered_by_currency_category)} операций по категории {category}") 

125 return filtered_by_currency_category 

126 else: 

127 reports_logger.warning(f"По категории {category} не найдено операций") 

128 return pd.DataFrame()

132def spending_by_weekday(transactions: pd.DataFrame, date: Optional[str] = None) -> pd.DataFrame: 

133 """Функция принимает на вход: 

134 датафрейм с транзакциями, 

135 опциональную дату (Если дата не передана, то берется текущая дата) 

136 Функция возвращает средние траты в каждый из дней недели за последние три месяца (от переданной даты).""" 

137 

138 # Получение объекта datetime из входящей строки

140 end_date = datetime.datetime.today() 

141 else: 

142 try: 

143 end_date = datetime.datetime.strptime(date, "%d-%m-%Y") 

144 except (TypeError, ValueError) as e: 

145 reports_logger.error(f"Ошибка {e} чтения даты из аргументов функции, требуется dd-mm-YYYY") 

146 return pd.DataFrame() 

147 start_date = end_date - relativedelta(months=3) 

148 reports_logger.debug(

150 ) 

151 

152 filtered_by_date_df = filter_transactions_by_date(df=transactions, start_date=start_date, end_date=end_date) 

153 reports_logger.debug(f"Транзакции отфильтрованы по дате, выбрано {len(filtered_by_date_df)} записей")

156 cleaned_for_calc_df = filtered_by_date_df[

160 ]

162 if cleaned_for_calc_df.empty: 

163 reports_logger.warning("Не найдено подходящих транзакций")
